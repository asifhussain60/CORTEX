Invocation Examples

Example 1: Using GovernanceMerger
from cortex.infrastructure import GovernanceMerger
from cortex.tools import @mcp_tool

@mcp_tool(
  name="governance_merger",
  description="Merge 4-tier governance rules",
  category="governance"
)
def merge_governance():
  merger = GovernanceMerger()
  merged_rules = merger.load_and_merge()
  # Returns: UnifiedInstructionSet with Tier 0 rules winning conflicts
Example 2: Using TDD-Master
from cortex.orchestrators import TDDMaster

@mcp_tool(
  name="tdd_master",
  description="Execute RED→GREEN→REFACTOR cycle"
)
def implement_ac_id(ac_id: str):
  tdd = TDDMaster(ac_id)
  red_result = tdd.red_phase()
  green_result = tdd.green_phase()
  refactor_result = tdd.refactor_phase()
  # Returns: Evidence bundle with tests, code, audit trail
