## 1. cortex-brain.db (Brain State)

```sql
-- Knowledge Graph (Tier 2)
CREATE TABLE knowledge_nodes (
    node_id TEXT PRIMARY KEY,
    node_type TEXT NOT NULL,  -- 'concept', 'pattern', 'lesson'
    content TEXT NOT NULL,
    metadata JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    access_count INTEGER DEFAULT 0
);

CREATE TABLE knowledge_edges (
    edge_id TEXT PRIMARY KEY,
    source_node_id TEXT NOT NULL,
    target_node_id TEXT NOT NULL,
    relationship_type TEXT NOT NULL,  -- 'relates_to', 'causes', 'implements'
    weight REAL DEFAULT 1.0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (source_node_id) REFERENCES knowledge_nodes(node_id),
    FOREIGN KEY (target_node_id) REFERENCES knowledge_nodes(node_id)
);

-- Development Context (Tier 3)
CREATE TABLE dev_hotspots (
    file_path TEXT PRIMARY KEY,
    complexity_score INTEGER,
    change_frequency INTEGER,
    last_modified TIMESTAMP,
    owner TEXT,
    test_coverage REAL
);

CREATE TABLE lessons_learned (
    lesson_id TEXT PRIMARY KEY,
    category TEXT NOT NULL,  -- 'architecture', 'testing', 'refactoring'
    title TEXT NOT NULL,
    description TEXT,
    example_code TEXT,
    tags JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Brain Protection (Tier 0)
CREATE TABLE skull_violations (
    violation_id TEXT PRIMARY KEY,
    rule_name TEXT NOT NULL,
    severity TEXT NOT NULL,  -- 'CRITICAL', 'HIGH', 'MEDIUM', 'LOW'
    file_path TEXT,
    line_number INTEGER,
    description TEXT,
    detected_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resolved_at TIMESTAMP
);

-- Indexes for performance
CREATE INDEX idx_nodes_type ON knowledge_nodes(node_type);
CREATE INDEX idx_edges_source ON knowledge_edges(source_node_id);
CREATE INDEX idx_edges_target ON knowledge_edges(target_node_id);
CREATE INDEX idx_hotspots_complexity ON dev_hotspots(complexity_score);
CREATE INDEX idx_lessons_category ON lessons_learned(category);
```

## 2. conversation-history.db (Tier 1 - Working Memory)

```sql
-- Conversation Storage (70-conv FIFO)
CREATE TABLE conversations (
    conv_id TEXT PRIMARY KEY,
    user_id TEXT NOT NULL,
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ended_at TIMESTAMP,
    total_messages INTEGER DEFAULT 0,
    metadata JSON
);

CREATE TABLE messages (
    message_id TEXT PRIMARY KEY,
    conv_id TEXT NOT NULL,
    role TEXT NOT NULL,  -- 'user', 'assistant', 'system'
    content TEXT NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    token_count INTEGER,
    intent TEXT,  -- Classified intent
    FOREIGN KEY (conv_id) REFERENCES conversations(conv_id)
);

CREATE TABLE context_snapshots (
    snapshot_id TEXT PRIMARY KEY,
    conv_id TEXT NOT NULL,
    file_context JSON,  -- Files in context
    brain_state JSON,   -- Active brain state
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (conv_id) REFERENCES conversations(conv_id)
);

-- FIFO Management
CREATE TABLE memory_metadata (
    key TEXT PRIMARY KEY,
    value TEXT,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Indexes
CREATE INDEX idx_messages_conv ON messages(conv_id);
CREATE INDEX idx_messages_timestamp ON messages(timestamp);
CREATE INDEX idx_conversations_started ON conversations(started_at);
```

## 3. cortex_metrics.db (Performance & Telemetry)

```sql
-- System Performance
CREATE TABLE performance_metrics (
    metric_id TEXT PRIMARY KEY,
    operation_name TEXT NOT NULL,

    memory_mb REAL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

-- Operation Tracking
CREATE TABLE operation_logs (
    log_id TEXT PRIMARY KEY,
    operation_type TEXT NOT NULL,  -- 'align', 'optimize', 'planning'

    duration_seconds REAL,

CREATE TABLE test_runs (
    run_id TEXT PRIMARY KEY,
    total_tests INTEGER,
    passed INTEGER,
    failed INTEGER,
    skipped INTEGER,
    duration_seconds REAL,
    pass_rate REAL,

CREATE TABLE complexity_history (
    record_id TEXT PRIMARY KEY,

CREATE INDEX idx_perf_operation ON performance_metrics(operation_name);

CREATE INDEX idx_ops_type ON operation_logs(operation_type);
CREATE INDEX idx_ops_status ON operation_logs(status);

```

## 4. cortex_alerts.db (System Alerts)

```sql
-- Alert Management
CREATE TABLE alerts (
    alert_id TEXT PRIMARY KEY,
    severity TEXT NOT NULL,  -- 'CRITICAL', 'WARNING', 'INFO'
    category TEXT NOT NULL,  -- 'system', 'test', 'security', 'performance'
    title TEXT NOT NULL,
    description TEXT,
    source TEXT,  -- Component that generated alert
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    acknowledged_at TIMESTAMP,
    resolved_at TIMESTAMP,
    metadata JSON
);

-- Alert Rules
CREATE TABLE alert_rules (
    rule_id TEXT PRIMARY KEY,
    rule_name TEXT NOT NULL,
    condition TEXT NOT NULL,  -- SQL-like condition
    severity TEXT NOT NULL,
    enabled BOOLEAN DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Indexes
CREATE INDEX idx_alerts_severity ON alerts(severity);
CREATE INDEX idx_alerts_created ON alerts(created_at);
CREATE INDEX idx_alerts_resolved ON alerts(resolved_at);
```

## 5. cortex_status.db (System Health)

```sql
-- System Status
CREATE TABLE system_status (
    status_id TEXT PRIMARY KEY,

    change_id TEXT PRIMARY KEY,
    config_key TEXT NOT NULL,
    old_value TEXT,
    new_value TEXT,
    changed_by TEXT,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Indexes
CREATE INDEX idx_status_component ON system_status(component);
CREATE INDEX idx_status_timestamp ON system_status(timestamp);
CREATE INDEX idx_health_timestamp ON healthcheck_history(timestamp);
```

## Database Access Patterns

### High-Frequency Operations
- Insert message (Tier 1): < 10ms
- Query knowledge node (Tier 2): < 50ms
- Record metric: < 5ms
- Check alert: < 20ms

### Batch Operations
- FIFO cleanup (70+ conversations): < 100ms
- Vacuum databases: < 5s
- Export knowledge graph: < 30s

### Maintenance Operations
- Index rebuild: < 60s
- Schema migration: < 120s
- Backup: < 10s per database

## Migration Strategy

```python
def migrate_database(db_path: str, target_version: int):
    """Apply incremental migrations to reach target version."""
    current = get_current_version(db_path)
    
    for version in range(current + 1, target_version + 1):
        migration_file = f"migrations/{version:03d}_migration.sql"
        apply_migration(db_path, migration_file)
        update_schema_version(db_path, version)
```

## Backup Strategy

**Automated Backups:**
- Daily: All databases → `cortex-brain/backups/daily/`
- Weekly: Full snapshot → `cortex-brain/backups/weekly/`
- Before major operations: Point-in-time snapshot

**Backup Command:**
```bash
sqlite3 cortex-brain.db ".backup 'cortex-brain/backups/cortex-brain_$(date +%Y%m%d).db'"
```

## Key Benefits

1. **Separation of Concerns:** Logical database separation by domain
2. **Performance:** Optimized indexes for common queries
3. **Scalability:** FIFO management prevents unbounded growth
4. **Maintainability:** Clear schema versioning and migrations
5. **Backup Safety:** Regular automated backups
6. **Query Efficiency:** < 100ms for 99% of operations
