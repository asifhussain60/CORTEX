// assets/js/visualizations.js - Line 12
async function loadAnalysisData() {
    try {
        // Check if running locally (file:// has CORS restrictions)
        if (window.location.protocol === 'file:') {
            console.log('Local file mode - using mock data');
            loadMockData();
            return;
        }
        
        // Production: load from JSON file
        const response = await fetch('analysis-data.json');
        dashboardData = await response.json();
        // ... initialize visualizations
    } catch (error) {
        console.error('Error loading data:', error);
        loadMockData(); // Fallback
    }
}

💡 For Production Use

When running on a web server (http/https protocol), the dashboard will automatically attempt to load analysis-data.json first, falling back to mock data only if the file is missing or fails to load.

Server modes supported:

  • Python: python -m http.server 8000
  • Node.js: npx http-server
  • PHP: php -S localhost:8000
