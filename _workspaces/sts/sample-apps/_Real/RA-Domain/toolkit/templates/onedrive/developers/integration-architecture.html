Purpose: Publish NServiceBus event when balance changes due to rollover
Business Logic: Publishes BalanceChangedEvent to message bus for downstream systems (statements, reporting, analytics) 

// Pseudocode based on AST analysis
async Task PublishBalanceChangedById(BalanceChangeInfoDto balanceChange)
{
    var @event = new BalanceChangedEvent
    {
        ReimbursementAccountId = balanceChange.AccountId,
        OldBalance = balanceChange.OldBalance,
        NewBalance = balanceChange.NewBalance,
        CarryoverAmount = balanceChange.CarryoverAmount,
        ForfeitedAmount = balanceChange.ForfeitedAmount,
        TransactionId = GetTransactionId(balanceChange),
        PlanYear = balanceChange.PlanYear,
        Timestamp = DateTime.UtcNow
    };

    await messageSession.Publish(@event);
}
