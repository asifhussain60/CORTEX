# Python Audit Logging for CORTEX Operations
from functools import wraps
from cortex_brain.tier0.audit import AuditLogger
import hashlib
import json

audit_logger = AuditLogger()

def audit_log(event_type: str, event_category: str, severity: str = 'info'):
    """Decorator for automatic audit logging of function calls."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Capture function entry
            event_data = {
                'function': func.__name__,
                'args': str(args),
                'kwargs': str(kwargs),
                'user': get_current_user(),
                'timestamp': datetime.utcnow().isoformat()
            }
            
            try:
                # Execute function
                result = func(*args, **kwargs)
                
                # Log success event
                audit_logger.log_event(
                    event_type=event_type,
                    event_category=event_category,
                    severity=severity,
                    actor=event_data['user'],
                    result='success',
                    context=event_data
                )
                
                return result
                
            except Exception as e:
                # Log failure event
                audit_logger.log_event(
                    event_type=event_type,
                    event_category=f"{event_category}_failed",
                    severity='error',
                    actor=event_data['user'],
                    result='failure',
                    error=str(e),
                    context=event_data
                )
                raise
        
        return wrapper
    return decorator
