The Testing Pyramid

Good test suites follow the testing pyramid: many fast unit tests at the base, fewer integration tests in the middle, and minimal E2E tests at the top.

Unit Tests (70%)

Test individual functions/classes in isolation. Fast, focused, run in milliseconds.

Integration Tests (20%)

Test how components work together. Database, API, service interactions.

E2E Tests (10%)

Test full user flows. Slow but verify real user scenarios work.

TDD Benefits

Write tests first, then code. Results in better design and fewer bugs.

// discount.service.ts - No tests!
export class DiscountService {
    calculateDiscount(
        order: Order, 
        user: User, 
        coupon?: Coupon
    ): number {
        let discount = 0;
        
        // Premium user discount
        if (user.tier === 'premium') {
            discount += order.total * 0.1;
        } else if (user.tier === 'gold') {
            discount += order.total * 0.15;
        }
        
        // Coupon discount
        if (coupon?.isValid) {
            if (coupon.type === 'percentage') {
                discount += order.total * (coupon.value / 100);
            } else {
                discount += coupon.value;
            }
        }
        
        // Cap discount at 50% of order
        return Math.min(discount, order.total * 0.5);
    }
}

// No tests = no confidence in:
// - Edge cases (negative values, null inputs)
// - Discount calculations are correct
// - Cap logic works properly
// - Refactoring won't break anything

// Interfaces for all dependencies
public interface IOrderRepository { Task SaveAsync(Order order); }
public interface IEmailService { Task SendAsync(string to, string msg); }
public interface IPaymentGateway { Task<PaymentResult> ChargeAsync(decimal amount); }
public interface ITaxCalculator { decimal GetRate(string state); }

public class OrderProcessor : IOrderProcessor
{
    private readonly IOrderRepository _repository;
    private readonly IEmailService _emailService;
    private readonly IPaymentGateway _paymentGateway;
    private readonly ITaxCalculator _taxCalculator;
    private readonly ILogger<OrderProcessor> _logger;
    
    // Dependencies injected - fully mockable!
    public OrderProcessor(
        IOrderRepository repository,
        IEmailService emailService,
        IPaymentGateway paymentGateway,
        ITaxCalculator taxCalculator,
        ILogger<OrderProcessor> logger)
    {
        _repository = repository;
        _emailService = emailService;
        _paymentGateway = paymentGateway;
        _taxCalculator = taxCalculator;
        _logger = logger;
    }
    
    public async Task ProcessOrderAsync(Order order)
    {
        _logger.LogInformation("Processing order {OrderId}", order.Id);
        
        order.Tax = order.Subtotal * _taxCalculator.GetRate(order.ShippingState);
        
        await _paymentGateway.ChargeAsync(order.Total);
        await _repository.SaveAsync(order);
        await _emailService.SendAsync(order.CustomerEmail, "Order confirmed!");
    }
}

Testing Best Practices

Testing Pyramid

70% unit tests, 20% integration, 10% E2E. Fast feedback at the base.

Unit → Integration → E2E

Arrange-Act-Assert

Structure tests clearly: setup, execute, verify. One assertion per test.

// Arrange // Act // Assert

Mock External Dependencies

Isolate units by mocking databases, APIs, and services.

Mock.Of<IRepository>()

Test-Driven Development

Red → Green → Refactor. Write failing test first, then make it pass.

expect().toBe() // fails first
