DRY - Don't Repeat Yourself

Angular

How This Was Refactored

  • Problem: Same validation logic duplicated across multiple components
  • Fix: Extract common validation into reusable service and directive
  • Result: Single source of truth - changes propagate automatically
Before: Duplicated Logic
// user-form.component.ts

  const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return regex.test(email);
}

// contact-form.component.ts
validateEmail(email: string): boolean {
  const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

KISS - Keep It Simple

TypeScript

How This Was Refactored

  • Problem: Over-engineered solution with unnecessary complexity
  • Fix: Use simple, direct approach - leverage built-in methods
  • Result: Readable, maintainable code that does exactly what's needed
Before: Over-Engineered
class ArrayProcessor<T> {
  private items: T[] = [];
  private filterFn: ((item: T) => boolean) | null = null;
  
  setItems(items: T[]): this { this.items = items; return this; }
  setFilter(fn: (item: T) => boolean): this { this.filterFn = fn; return this; }
  
  execute(): T[] {
    if (!this.filterFn) throw new Error('Filter required');
    return this.items.filter(this.filterFn);
  }
}

// Usage
const result = new ArrayProcessor<number>()
  .setItems([1, 2, 3, 4, 5])
  .setFilter(n => n > 2)
  .execute();
After: Simple & Direct
// Just use the built-in array method!
const numbers = [1, 2, 3, 4, 5];
const result = numbers.filter(n => n > 2);

// For reusability, a simple function is enough
function filterGreaterThan<T extends number>(
  items: T[], 
  threshold: T
): T[] {
  return items.filter(item => item > threshold);
}

const filtered = filterGreaterThan(numbers, 2);

YAGNI - You Aren't Gonna Need It

C#

How This Was Refactored

  • Problem: Premature abstractions and features "just in case"
  • Fix: Remove unused code - implement only what's actually needed now
  • Result: Lean codebase that's easier to understand and maintain
Before: Premature Features
public interface IConfigProvider { }
public interface IConfigProviderV2 : IConfigProvider { }
public interface IConfigFactory { }

public class ConfigManager
{
    private IConfigProvider _provider;
    private IConfigFactory _factory; // Never used
    private bool _enableCaching = false; // Never used
    private Dictionary<string, object> _cache; // Never used
    
    public string GetSetting(string key) => 
        Environment.GetEnvironmentVariable(key);
    
    // "Might need this later"
    public void SetSetting(string key, string value) { }
    public void RefreshCache() { }
    public void EnableDistributedMode() { }
}
After: Only What's Needed
public class ConfigService
{
    public string GetSetting(string key) =>
        Environment.GetEnvironmentVariable(key) 
        ?? throw new KeyNotFoundException(key);
    
    public T GetSetting<T>(string key, T defaultValue = default)
    {
        var value = Environment.GetEnvironmentVariable(key);
        if (string.IsNullOrEmpty(value)) return defaultValue;
        return (T)Convert.ChangeType(value, typeof(T));
    }
}
