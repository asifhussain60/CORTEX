What are Anti-Patterns?

Anti-patterns are common solutions to recurring problems that are ineffective and counterproductive. Unlike design patterns (good solutions), anti-patterns are bad practices that appear beneficial but create more problems than they solve.

Why Learn Anti-Patterns?

Recognition: Identify problematic code before it spreads
Prevention: Avoid making the same mistakes
Refactoring: Transform anti-patterns into clean code
Code Review: Spot anti-patterns during reviews

Source

Based on AntiPatterns: Refactoring Software, Architectures, and Projects in Crisis (Brown et al. 1998)

Pattern-Based Detection

REGEX PATTERNS

✅ Detection Patterns

import re
import ast

def detect_anti_patterns(code: str) -> list:
    """Detect anti-patterns using regex and AST analysis."""
    detected = []
    
    # Deep inheritance (class ... extends ... extends)
    if re.search(r'class.*extends.*extends.*extends', code):
        detected.append("Deep Inheritance Tree")
    
    # Complex conditionals (multiple else if)
    if re.search(r'if.*else if.*else if.*else if', code):
        detected.append("Complex Conditional")
    
    # Technical debt markers
    if re.search(r'TODO|FIXME|HACK|XXX', code):
        detected.append("Technical Debt Markers")
    
    # Singleton pattern overuse
    if re.search(r'getInstance\(\)', code):
        detected.append("Possible Singleton Overuse")
    
    return detected

# Example
code = """
class DatabaseManager:
    _instance = None
    
    @staticmethod
    def getInstance():
        if DatabaseManager._instance is None:
            DatabaseManager._instance = DatabaseManager()
        return DatabaseManager._instance
    
    # TODO: Fix this hack
    def query(self, sql):
        pass  # FIXME: No error handling
"""

issues = detect_anti_patterns(code)
print(issues)  # ['Technical Debt Markers', 'Possible Singleton Overuse']

God Object (The Blob)

CRITICAL SEVERITY

Description

A single class that knows too much or does too much, taking on most responsibilities in the system. Results in tight coupling and low cohesion.

Symptoms

Single class with >30 methods
Class with >1000 lines of code
Generic class name (Manager, Controller, Handler, Service)
Many dependencies on this class
Class touches many different concerns
Difficult to test in isolation

❌ God Object Anti-Pattern

# ❌ God Object
class UserManager:
    """Single class doing everything user-related."""
    
    def authenticate(self, username, password):
        """Authentication logic."""
        # Database access
        user = self.db.query("SELECT * FROM users WHERE username = ?", username)
        # Password hashing
        hashed = self.hash_password(password)
        # Validation
        return user.password == hashed
    
    def authorize(self, user, resource):
        """Authorization logic."""
        # RBAC logic
        roles = self.db.query("SELECT role FROM user_roles WHERE user_id = ?", user.id)
        return resource.required_role in roles
    
    def send_email(self, user, message):
        """Email sending logic."""
        # SMTP configuration
        smtp = self.get_smtp_connection()
        smtp.send(user.email, message)
    
    def send_sms(self, user, message):
        """SMS sending logic."""
        # Twilio API
        self.twilio_client.send(user.phone, message)
    
    def generate_report(self, user):
        """Report generation."""
        # PDF generation
        pdf = PDFGenerator()
        pdf.add_user_data(user)
        return pdf.render()
    
    def update_profile(self, user, data):
        """Profile update."""
        # Validation
        self.validate_profile_data(data)
        # Database update
        self.db.update("users", user.id, data)
        # Cache invalidation
        self.cache.delete(f"user:{user.id}")
    
    # ... 25+ more methods doing everything

✅ Refactored: Single Responsibility

# ✅ Separated responsibilities
class AuthenticationService:
    """Only handles authentication."""
    
    def __init__(self, user_repository, password_hasher):
        self.users = user_repository
        self.hasher = password_hasher
    
    def authenticate(self, username, password):
        user = self.users.find_by_username(username)
        if not user:
            return None
        hashed = self.hasher.hash(password)
        return user if user.password == hashed else None

class AuthorizationService:
    """Only handles authorization."""
    
    def __init__(self, role_repository):
        self.roles = role_repository
    
    def authorize(self, user, resource):
        user_roles = self.roles.get_user_roles(user.id)
        return resource.required_role in user_roles

class NotificationService:
    """Only handles notifications."""
    
    def __init__(self, email_sender, sms_sender):
        self.email = email_sender
        self.sms = sms_sender
    
    def send_email(self, user, message):
        self.email.send(user.email, message)
    
    def send_sms(self, user, message):
        self.sms.send(user.phone, message)

class UserProfileService:
    """Only handles user profiles."""
    
    def __init__(self, user_repository, cache):
        self.users = user_repository
        self.cache = cache
    
    def update_profile(self, user, data):
        self.users.update(user.id, data)
        self.cache.delete(f"user:{user.id}")

# Now each class has ONE reason to change (Single Responsibility)

Refactoring Strategy

Identify distinct responsibilities
Create separate classes for each responsibility
Use dependency injection to connect them
Apply Single Responsibility Principle

Spaghetti Code

CRITICAL SEVERITY

Description

Code with complex and tangled control structures, making it nearly impossible to follow the logic. Often results from lack of proper design.

Symptoms

Excessive use of GOTO or early returns
Deeply nested conditionals (5+ levels)
Long methods with many branches
Global state mutations throughout
No clear flow or structure

❌ Spaghetti Code

# ❌ Spaghetti Code
def process_order(order):
    if order.status == "pending":
        if order.payment_method == "credit_card":
            if order.card_valid():
                if order.inventory_check():
                    if order.fraud_check():
                        order.charge_card()
                        order.update_inventory()
                        order.status = "processing"
                        if order.is_express:
                            order.ship_express()
                            if order.notification_enabled:
                                send_email(order.customer)
                        else:
                            order.ship_standard()
                    else:
                        order.status = "fraud_review"
                        notify_fraud_team(order)
                else:
                    order.status = "out_of_stock"
                    send_backorder_email(order.customer)
            else:
                order.status = "payment_failed"
                send_payment_failed_email(order.customer)
        elif order.payment_method == "paypal":
            # Another 30 lines of nested ifs...
            pass
    elif order.status == "processing":
        # More nested logic...
        pass
    # ... continues for 200+ lines

✅ Refactored: Clear Structure

# ✅ Clean, structured code
class OrderProcessor:
    """Process orders with clear steps."""
    
    def process(self, order):
        """Main processing pipeline."""
        if not self.validate_order(order):
            return False
        
        if not self.process_payment(order):
            return False
        
        if not self.check_inventory(order):
            return False
        
        self.fulfill_order(order)
        self.send_notifications(order)
        return True
    
    def validate_order(self, order):
        """Validate order can be processed."""
        if order.status != "pending":
            return False
        
        if not order.fraud_check():
            self.handle_fraud(order)
            return False
        
        return True
    
    def process_payment(self, order):
        """Process payment based on method."""
        payment_processor = PaymentProcessorFactory.create(order.payment_method)
        
        try:
            payment_processor.charge(order)
            return True
        except PaymentError as e:
            self.handle_payment_failure(order, e)
            return False
    
    def check_inventory(self, order):
        """Check and reserve inventory."""
        if not order.inventory_check():
            self.handle_out_of_stock(order)
            return False
        
        order.update_inventory()
        return True
    
    def fulfill_order(self, order):
        """Fulfill the order."""
        order.status = "processing"
        
        shipper = ShipperFactory.create(
            "express" if order.is_express else "standard"
        )
        shipper.ship(order)
    
    def send_notifications(self, order):
        """Send customer notifications."""
        if order.notification_enabled:
            NotificationService.send_order_confirmation(order.customer, order)

# Each method does ONE thing and is easy to understand

Deep Inheritance Tree

HIGH SEVERITY

Description

Inheritance hierarchy with too many levels (>5), making code hard to understand and maintain. Violates "favor composition over inheritance."

❌ Deep Inheritance

# ❌ Deep inheritance (7 levels!)
class Vehicle: pass
class MotorVehicle(Vehicle): pass
class PassengerVehicle(MotorVehicle): pass
class Car(PassengerVehicle): pass
class Sedan(Car): pass
class LuxurySedan(Sedan): pass
class ElectricLuxurySedan(LuxurySedan): pass  # Too deep!

✅ Refactored: Composition

# ✅ Composition over inheritance
class Vehicle:
    def __init__(self, engine, body_style, features):
        self.engine = engine          # Composition
        self.body_style = body_style  # Composition
        self.features = features      # Composition

# Engines
class ElectricEngine: pass
class GasEngine: pass

# Body styles
class SedanBody: pass
class SUVBody: pass

# Features
class LuxuryFeatures: pass
class StandardFeatures: pass

# Create vehicle with composition
luxury_electric_sedan = Vehicle(
    engine=ElectricEngine(),
    body_style=SedanBody(),
    features=LuxuryFeatures()
)
# Flat hierarchy, flexible composition

Singleton Overuse

MEDIUM SEVERITY

Description

Overusing the Singleton pattern creates global state, tight coupling, and testability issues. Singletons are essentially global variables in disguise.

When Singletons Are Acceptable

✅ Logging
✅ Configuration (read-only)
✅ Hardware interfaces (printer, serial port)
❌ Database connections (use connection pool)
❌ Business logic
❌ Most other cases

❌ Singleton Overuse

# ❌ Everything is a Singleton!
class DatabaseManager:  # Singleton
    _instance = None
    @staticmethod
    def getInstance(): ...

class UserService:  # Singleton
    _instance = None
    @staticmethod
    def getInstance(): ...

class EmailService:  # Singleton
    _instance = None
    @staticmethod
    def getInstance(): ...

# Problems:
# - Hard to test (can't mock)
# - Hidden dependencies
# - Global state
# - Tight coupling

✅ Dependency Injection Instead

# ✅ Dependency Injection
class UserService:
    """No singleton! Dependencies injected."""
    
    def __init__(self, database, email_service):
        self.db = database
        self.email = email_service
    
    def create_user(self, username, email):
        user = self.db.insert_user(username, email)
        self.email.send_welcome(user)
        return user

# Easy to test with mocks
def test_create_user():
    mock_db = MockDatabase()
    mock_email = MockEmailService()
    service = UserService(mock_db, mock_email)
    
    user = service.create_user("john", "john@example.com")
    
    assert mock_db.insert_user_called
    assert mock_email.send_welcome_called

Copy-Paste Programming

HIGH SEVERITY

Description

Duplicating code by copying and pasting instead of creating reusable abstractions. Violates DRY (Don't Repeat Yourself).

❌ Code Duplication

# ❌ Copy-pasted code
def create_user(username, email):
    if not username or len(username) < 3:
        raise ValueError("Invalid username")
    if not email or "@" not in email:
        raise ValueError("Invalid email")
    user = db.insert("users", {"username": username, "email": email})
    log.info(f"Created user: {username}")
    return user

def update_user(user_id, username, email):
    # Same validation copied!
    if not username or len(username) < 3:
        raise ValueError("Invalid username")
    if not email or "@" not in email:
        raise ValueError("Invalid email")
    user = db.update("users", user_id, {"username": username, "email": email})
    log.info(f"Updated user: {username}")
    return user

# If validation rules change, must update in 2+ places!

✅ DRY Refactoring

# ✅ Reusable validation
def validate_username(username):
    """Validate username (single source of truth)."""
    if not username or len(username) < 3:
        raise ValueError("Invalid username")

def validate_email(email):
    """Validate email (single source of truth)."""
    if not email or "@" not in email:
        raise ValueError("Invalid email")

def create_user(username, email):
    validate_username(username)  # Reuse
    validate_email(email)        # Reuse
    user = db.insert("users", {"username": username, "email": email})
    log.info(f"Created user: {username}")
    return user

def update_user(user_id, username, email):
    validate_username(username)  # Reuse
    validate_email(email)        # Reuse
    user = db.update("users", user_id, {"username": username, "email": email})
    log.info(f"Updated user: {username}")
    return user

# Validation rules in ONE place. Easy to change.
