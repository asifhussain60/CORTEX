Domain RAG Integration

Domain RAG Integration connects retrieval systems with domain-specific knowledge, business rules, and organizational context. This enables AI systems to provide contextually relevant responses aligned with domain expertise.

Key Concepts

Domain Knowledge Sources: Engineering practices, coding standards, design patterns
Context-Aware Retrieval: Retrieve based on domain context (code review, generation, architecture)
Multi-Tenant Isolation: Separate knowledge bases per tenant/organization
Hierarchical Knowledge: General patterns → Domain-specific → Organization-specific

CORTEX Brain Integration

CORTEX uses a 4-tier brain architecture. RAG integrates with each tier for different purposes.

Tier 0: Governance & Rules

CRITICAL

Retrieve SKULL rules, brain protection rules, and governance policies.

✅ Governance Retrieval

def retrieve_governance_rules(operation: str, vector_db):
    """
    Retrieve SKULL rules relevant to the operation.
    """
    query = f"What rules apply when {operation}?"
    
    rules = vector_db.search(
        query=query,
        filter={"tier": "tier0", "type": "governance"},

Code Generation Pattern

COMMON USE CASE

Retrieve engineering patterns, coding standards, and similar code before generation.

✅ Code Generation with RAG

def generate_code_with_rag(user_request: str, vector_db, llm):
    """
    Generate code using retrieved patterns and standards.
    """
    # Step 1: Retrieve relevant patterns
    patterns = vector_db.search(
        query=user_request,
        filter={"category": "engineering", "type": "pattern"},
        top_k=3
    )
    
    # Step 2: Retrieve coding standards
    standards = vector_db.search(
        query=f"{user_request} coding standards",
        filter={"type": "standards"},
        top_k=2
    )
    
    # Step 3: Retrieve similar code examples
    examples = vector_db.search(
        query=user_request,
        filter={"type": "code_example"},
        top_k=2
    )
    
    # Step 4: Assemble context
    context = f"""
Engineering Patterns:
{format_patterns(patterns)}

Coding Standards:
{format_standards(standards)}

Similar Examples:
{format_examples(examples)}
"""
    
    # Step 5: Generate code
    prompt = f"""Based on these patterns and standards:
{context}

Generate code for: {user_request}

Requirements:
- Follow coding standards
- Apply relevant patterns
- Include error handling
- Add docstrings
"""
    
    code = llm.generate(prompt)
    return code

# Example
code = generate_code_with_rag(
    user_request="Create a REST API endpoint for user registration",
    vector_db=knowledge_db,
    llm=openai_llm
)
# Returns: Code following patterns (Repository, DTO), standards (PEP 8), 
#          with examples (FastAPI endpoint structure)

def review_code_with_rag(code: str, language: str, vector_db, llm):
    """
    Review code using retrieved checklists and patterns.
    """
    # Step 1: Retrieve review checklist
    checklist = vector_db.search(
        query=f"{language} code review checklist",
        filter={"type": "checklist"},
        top_k=1
    )[0]
    
    # Step 2: Retrieve security patterns (OWASP)
    security = vector_db.search(
        query=f"{language} security vulnerabilities",
        filter={"category": "security"},
        top_k=5
    )
    
    # Step 3: Retrieve SOLID principles
    principles = vector_db.search(
        query="SOLID principles",
        filter={"category": "engineering"},
        top_k=1
    )[0]
    
    # Step 4: Perform review
    prompt = f"""Review this {language} code:

{code}

Check against:
1. Code Review Checklist: {checklist.content}
2. Security Patterns: {format_patterns(security)}
3. SOLID Principles: {principles.content}

Provide:
- Issues found (with severity)
- Suggestions for improvement
- Security vulnerabilities
"""
    
    review = llm.generate(prompt)
    return review

# Example
review = review_code_with_rag(
    code=user_submitted_code,
    language="Python",
    vector_db=knowledge_db,
    llm=openai_llm
)
# Returns: Detailed review against checklist, security patterns, SOLID principles

def architecture_advice_with_rag(scenario: str, vector_db, llm):
    """
    Provide architecture advice using retrieved patterns.
    """
    # Step 1: Retrieve relevant patterns
    patterns = vector_db.search(
        query=scenario,
        filter={"category": "architecture"},
        top_k=5
    )
    
    # Step 2: Retrieve anti-patterns to avoid
    anti_patterns = vector_db.search(
        query=f"{scenario} anti-patterns",
        filter={"type": "anti-pattern"},
        top_k=3
    )
    
    # Step 3: Generate advice
    prompt = f"""Given this scenario: {scenario}

Relevant Patterns:
{format_patterns(patterns)}

Anti-Patterns to Avoid:
{format_patterns(anti_patterns)}

Provide architectural advice including:
- Recommended patterns and why
- Trade-offs to consider
- Anti-patterns to avoid
- Implementation guidance
"""
    
    advice = llm.generate(prompt)
    return advice

# Example
advice = architecture_advice_with_rag(
    scenario="Building a microservices system with event-driven communication",
    vector_db=knowledge_db,
    llm=openai_llm
)
# Returns: Event sourcing, CQRS, Saga patterns; avoids distributed monolith

Multi-Tenant RAG

Namespace-Based Isolation

SECURITY

Isolate tenant data using metadata filters.

✅ Multi-Tenant Retrieval

def multi_tenant_retrieval(query: str, tenant_id: str, vector_db, top_k=5):
    """
    Retrieve knowledge scoped to specific tenant.
    """
    # Combine shared + tenant-specific knowledge
    
    # 1. Shared knowledge (all tenants)
    shared_results = vector_db.search(
        query=query,
        filter={"tenant_id": "shared", "type": "general_knowledge"},
        top_k=top_k // 2
    )
    
    # 2. Tenant-specific knowledge
    tenant_results = vector_db.search(
        query=query,
        filter={"tenant_id": tenant_id},
        top_k=top_k // 2
    )
    
    # 3. Merge and rank
    all_results = shared_results + tenant_results
    all_results.sort(key=lambda x: x.score, reverse=True)
    
    return all_results[:top_k]

# Example
results = multi_tenant_retrieval(
    query="Python best practices",
    tenant_id="company_abc",
    vector_db=knowledge_db,
    top_k=5
)
# Returns: 
# - Shared: General Python best practices (PEP 8, type hints)
# - Tenant: Company ABC's specific coding standards and internal libraries

Knowledge Hierarchy

Shared: General best practices, language patterns
Tenant-Specific: Company standards, internal APIs, project context
Isolation: Tenant A cannot access Tenant B's knowledge

Evaluation Framework

Quality Metrics

CONTINUOUS IMPROVEMENT

Track metrics to improve RAG system quality over time.

Key Metrics

Retrieval Precision: % of retrieved docs that are relevant (target: > 80%)
Retrieval Recall: % of relevant docs that are retrieved (target: > 70%)
Response Accuracy: Correctness of generated answers (human eval)
Citation Correctness: Citations match retrieved sources (target: > 95%)
Latency: End-to-end response time (target: < 3s)

✅ Metrics Collection

class RAGMetrics:
    """Collect and track RAG system metrics."""
    
    def __init__(self, metrics_db):
        self.metrics_db = metrics_db
    
    def log_retrieval(self, query: str, results: list, relevance_scores: list):
        """Log retrieval metrics."""
        self.metrics_db.insert({
            "timestamp": datetime.now(),
            "type": "retrieval",
            "query": query,
            "num_results": len(results),
            "avg_relevance": np.mean(relevance_scores),
            "latency_ms": results.metadata["latency_ms"]
        })
    
    def log_response(self, query: str, response: str, user_feedback: int):
        """Log response quality (1-5 stars)."""
        self.metrics_db.insert({
            "timestamp": datetime.now(),
            "type": "response",
            "query": query,
            "response_length": len(response),
            "user_rating": user_feedback
        })
    
    def compute_metrics(self, time_window="7d"):
        """Compute aggregated metrics."""
        metrics = self.metrics_db.aggregate(time_window)
        return {
            "avg_retrieval_latency": metrics["avg_latency_ms"],
            "avg_relevance_score": metrics["avg_relevance"],
            "avg_user_rating": metrics["avg_user_rating"],
            "total_queries": metrics["count"]
        }

# Usage
metrics = RAGMetrics(metrics_db)

# Log each interaction
results = vector_db.search(query, top_k=5)
metrics.log_retrieval(query, results, [r.score for r in results])

response = llm.generate(query, context)
user_feedback = collect_feedback()  # 1-5 stars
metrics.log_response(query, response, user_feedback)

# Weekly review
weekly_metrics = metrics.compute_metrics(time_window="7d")
print(f"Avg user rating: {weekly_metrics['avg_user_rating']}/5")
