Key Messaging Patterns

📤

Publish-Subscribe Pattern

HIGH

Description: Publishers send messages to topics without knowing subscribers. Subscribers receive all messages from subscribed topics, enabling loose coupling.

✅ Kafka Producer

from kafka import KafkaProducer
import json

producer = KafkaProducer(
    bootstrap_servers=['localhost:9092'],
    value_serializer=lambda v: json.dumps(v).encode('utf-8'),
    acks='all',  # Wait for all replicas
    retries=3
)

# Publish event
event = {'user_id': 123, 'action': 'order_placed'}
producer.send('orders', value=event, key=b'user-123')
producer.flush()

✅ Kafka Consumer

from kafka import KafkaConsumer
import json

consumer = KafkaConsumer(
    'orders',
    bootstrap_servers=['localhost:9092'],
    group_id='order-processor',
    auto_offset_reset='earliest',
    value_deserializer=lambda v: json.loads(v.decode('utf-8'))
)

for message in consumer:
    event = message.value
    process_order(event)
🔄

Event Sourcing Pattern

HIGH

Description: Store all changes as a sequence of events rather than just current state. Rebuild state by replaying events. Enables audit trails and temporal queries.

✅ Event Store Pattern

class Event:
    def __init__(self, event_type, data, timestamp=None):
        self.event_type = event_type
        self.data = data
        self.timestamp = timestamp or datetime.utcnow()

class EventStore:
    def __init__(self):
        self.events = []
    
    def append(self, stream_id, event):
        self.events.append({
            'stream_id': stream_id,
            'event': event,
            'version': len(self.events)
        })
    
    def get_events(self, stream_id):
        return [e['event'] for e in self.events 
                if e['stream_id'] == stream_id]

# Reconstruct state from events
def rebuild_order(events):
    order = Order()
    for event in events:
        order.apply(event)
    return order
⚖️

CQRS Pattern

HIGH

Description: Separate read and write models. Commands modify state, queries read from optimized projections. Enables independent scaling and optimization.

✅ CQRS Implementation

# Command Side (Write Model)
class CreateOrderCommand:
    def __init__(self, customer_id, items):
        self.customer_id = customer_id
        self.items = items

class OrderCommandHandler:
    def handle(self, command: CreateOrderCommand):
        order = Order.create(command.customer_id, command.items)
        self.event_store.append(order.id, OrderCreatedEvent(order))
        return order.id

# Query Side (Read Model)
class OrderReadModel:
    """Denormalized view optimized for reads"""
    def __init__(self, order_id, customer_name, total, status):
        self.order_id = order_id
        self.customer_name = customer_name
        self.total = total
        self.status = status

class OrderQueryService:
    def get_customer_orders(self, customer_id):
        return self.read_db.query(OrderReadModel)\
                          .filter_by(customer_id=customer_id)\
                          .all()

RabbitMQ Exchange Patterns

MEDIUM

Description: RabbitMQ supports different exchange types: Direct (exact routing), Fanout (broadcast), Topic (pattern matching), and Headers (attribute-based routing).

✅ Topic Exchange

import pika

connection = pika.BlockingConnection(
    pika.ConnectionParameters('localhost')
)
channel = connection.channel()

# Declare topic exchange
channel.exchange_declare(
    exchange='orders', 
    exchange_type='topic',
    durable=True
)

# Publish with routing key
channel.basic_publish(
    exchange='orders',
    routing_key='order.created.usa',
    body=json.dumps(order_data),
    properties=pika.BasicProperties(
        delivery_mode=2,  # Persistent
        content_type='application/json'
    )
)

# Subscribe to pattern
channel.queue_bind(
    exchange='orders',
    queue='usa_orders',
    routing_key='order.*.usa'
)

        self.processed = processed_events
    
    def handle(self, event: OrderCreatedEvent):
        # Idempotency check
        if event.event_id in self.processed:
            logger.info(f"Skipping duplicate: {event.event_id}")
            return
        
        try:
            self.process_order(event)
            self.processed.add(event.event_id)
        except Exception as e:
            # Send to dead letter queue

🔍 Code Review Integration

CORTEX reviews messaging code for common issues:

  • No Idempotency: Flags handlers without deduplication
