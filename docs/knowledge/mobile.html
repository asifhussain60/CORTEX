Description: All animations, scrolling, and UI interactions must maintain 60 FPS (16.67ms per frame). Heavy computations must be offloaded from the main/UI thread.

Rationale: Frame drops cause visible jank that degrades perceived quality. Users notice anything below 60 FPS.

    Jetpack Compose MVVM Example

    // ViewModel with StateFlow
class UserViewModel(
    private val repository: UserRepository
) : ViewModel() {
    
    private val _uiState = MutableStateFlow(UserUiState())
    val uiState: StateFlow<UserUiState> = _uiState.asStateFlow()
    
    init {
        loadUsers()
    }
    
    private fun loadUsers() {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            
            repository.getUsers()
                .catch { error ->
                    _uiState.update { 
                        it.copy(isLoading = false, error = error.message) 
                    }
                }
                .collect { users ->
                    _uiState.update { 
                        it.copy(isLoading = false, users = users) 
                    }
                }
        }
    }
}

    React Native Component Example

    // Optimized FlatList with TypeScript
import React, { useCallback, useMemo } from 'react';
import { FlatList, ListRenderItem, StyleSheet } from 'react-native';

interface User {
  id: string;
  name: string;
  email: string;
}

interface UserListProps {
  users: User[];
  onUserPress: (user: User) => void;
}

export const UserList: React.FC<UserListProps> = React.memo(({ 
  users, 
  onUserPress 
}) => {
  // Memoize keyExtractor
  const keyExtractor = useCallback((item: User) => item.id, []);
  
  // Memoize renderItem with useCallback
  const renderItem: ListRenderItem<User> = useCallback(({ item }) => (
    <UserItem 
      user={item} 
      onPress={onUserPress}
    />
  ), [onUserPress]);
  
  // Memoize getItemLayout for fixed-height items
  const getItemLayout = useMemo(() => (
    _: any, index: number
  ) => ({
    length: USER_ITEM_HEIGHT,
    offset: USER_ITEM_HEIGHT * index,
    index,
  }), []);

  return (
    <FlatList
      data={users}
      keyExtractor={keyExtractor}
      renderItem={renderItem}
      getItemLayout={getItemLayout}
      removeClippedSubviews={true}
      maxToRenderPerBatch={10}
      windowSize={5}
      initialNumToRender={10}
    />
  );
});

const USER_ITEM_HEIGHT = 72;

    Flutter with Riverpod Example

    // State management with Riverpod
import 'package:flutter_riverpod/flutter_riverpod.dart';

// Define providers
final userRepositoryProvider = Provider<UserRepository>((ref) {
  return UserRepository();
});

final usersProvider = FutureProvider<List<User>>((ref) async {
  final repository = ref.watch(userRepositoryProvider);
  return repository.fetchUsers();
});

// Stateless widget consuming provider
class UsersScreen extends ConsumerWidget {
  const UsersScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final usersAsync = ref.watch(usersProvider);

    return Scaffold(
      appBar: AppBar(title: const Text('Users')),
      body: usersAsync.when(
        data: (users) => ListView.builder(
          itemCount: users.length,
          itemBuilder: (context, index) {
            final user = users[index];
            return UserTile(user: user);
          },
        ),
        loading: () => const Center(
          child: CircularProgressIndicator(),
        ),
        error: (error, stack) => Center(
          child: Text('Error: $error'),
        ),
      ),
    );
  }
}

// Extracted stateless widget with const constructor
class UserTile extends StatelessWidget {
  const UserTile({super.key, required this.user});
  
  final User user;

  @override
  Widget build(BuildContext context) {
    return ListTile(
      leading: CircleAvatar(child: Text(user.name[0])),
      title: Text(user.name),
      subtitle: Text(user.email),
    );
  }
}

    Touch Target Guidelines

    • Minimum 44x44pt touch targets (iOS) - Apple HIG requirement
    • Minimum 48x48dp touch targets (Android) - Material Design spec
    • 8dp spacing between targets - Prevent accidental taps
    • Consider thumb zones - Primary actions in easy reach
