```python
import shutil
import json

def _generate_plan_viewer_html(self, plan_folder: Path, plan_data: Dict[str, Any]) -> None:
    """Generate HTML plan viewer using templates"""
    # 1. Copy template files
    template_dir = Path(__file__).parent.parent.parent / 'templates' / 'plan-viewer'
    
    if not template_dir.exists():
        self.logger.warning(f"Template directory not found: {template_dir}")
        return
    
    # Copy all template files to plan folder
    for file_name in ['plan-viewer.html', 'styles.css', 'viewer.js', 'README.md']:
        src_file = template_dir / file_name
        dst_file = plan_folder / file_name
        
        if src_file.exists():
            shutil.copy2(src_file, dst_file)
            self.logger.debug(f"Copied template: {file_name}")
    
    # 2. Generate plan-data.json
    plan_metadata = {
        'plan_id': plan_data.get('plan_id', 'UNKNOWN'),
        'plan_title': plan_data.get('plan_title', 'Untitled Plan'),
        'status': plan_data.get('status', 'not_started'),
        'estimated_hours': plan_data.get('estimated_hours', 0.0),
        'author': plan_data.get('author', 'CORTEX'),
        'created_at': plan_data.get('created_at', datetime.now().isoformat()),
        'description': plan_data.get('description', ''),
        'phases': [
            {
                'number': phase.get('number', i+1),
                'name': phase.get('name', f'Phase {i+1}'),
                'estimated_hours': phase.get('estimated_hours', 0.0),
                'description': phase.get('description', '')
            }
            for i, phase in enumerate(plan_data.get('phases', []))
        ],
        'technologies': plan_data.get('technologies', []),
        'deliverables': plan_data.get('deliverables', [])
    }
    
    data_file = plan_folder / 'plan-data.json'
    data_file.write_text(json.dumps(plan_metadata, indent=2), encoding='utf-8')
    
    self.logger.info(f"Generated plan viewer at: {plan_folder / 'plan-viewer.html'}")
    self.logger.info(f"Open in browser: file://{plan_folder / 'plan-viewer.html'}")
```

