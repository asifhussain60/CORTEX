Quick Start

# Basic Usage from core.toolkit_manager import ToolkitManager, ExecutionContext # Initialize manager = ToolkitManager() # Execute a tool (async) result = await manager.execute('align', ['--check-only']) # Check result if result.success: print(result.stdout) else: print(f"Error: {result.error}")
# With Dry Run & Checkpoint context = ExecutionContext( tool="cleanup", args=["--all"], dry_run=True, # Preview without executing checkpoint_enabled=True # Auto-checkpoint for rollback ) result = await manager.execute("cleanup", ["--all"], context)
